- ✅ **流动性评估** - 判断是否容易买卖

**实现方法：**
```python

def parse_log(log_file):
    coins = {}
    for line in log_file:
        if 'USDT:' in line and '%' in line:
            symbol, change, volume, market_cap = extract_data(line)
            coins[symbol].append({
                'change': change,
                'volume': volume,
                'market_cap': market_cap
            })
    return coins

# 2. 统计分析
def analyze_coins(coins):
    top10 = sorted(coins.items(),
                   key=lambda x: len(x[1]),
                   reverse=True)[:10]
    return top10

# 3. 生成报告
def generate_report(top10):
    for rank, (symbol, data) in enumerate(top10, 1):
        avg_change = mean([d['change'] for d in data])
        frequency = len(data)
        print(f"{rank}. {symbol}: 出现{frequency}次, 平均涨跌{avg_change:.2f}%")
```
